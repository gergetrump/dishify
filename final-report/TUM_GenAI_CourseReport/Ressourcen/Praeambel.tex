\newcommand{\screenshot}[2][\textwidth]{%
    \begingroup
    \setlength{\fboxsep}{0pt}%
    \setlength{\fboxrule}{0.4pt}%
    \fcolorbox{framegrey}{white}{%
        \includegraphics[width=\dimexpr#1-2\fboxrule\relax]{#2}}%
    \endgroup
}
